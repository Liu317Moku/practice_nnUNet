import sys

import nnunetv2

print("\nPython version")

print("\nUser site")

print(" ", torch.__version__)

print("\nTorch CUDA")

print("\nBatchgenerators location")
print(" ", batchgenerators.__file__)

print("\nExecutables")

for cmd in [
    "nnUNetv2_plan_and_preprocess",
    "nnUNetv2_extract_fingerprint",
    "nnUNetv2_plan_experiment",
    "nnUNetv2_preprocess",
    "nnUNetv2_train",
    "nnUNetv2_predict",
]:
    print(f" {cmd}: {shutil.which(cmd)}")

print("\nGCC")
print(" ", shutil.which("x86_64-conda-linux-gnu-gcc"))

print("\nG++")
print(" ", shutil.which("x86_64-conda-linux-gnu-g++"))

print("=" * 70)
